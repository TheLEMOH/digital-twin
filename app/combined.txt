transform.ts
export enum Indicators {
    temp = "temp",
    press = "prmsl",
    gust = "gust",
    ugrd = "ugrd",
    vgrd = "vgrd",
}

export const dataExtremes: { [x: string]: [number, number] } = {
    [Indicators.temp]: [-100, 70],
    [Indicators.press]: [85000 / 133.332, 109000 / 133.332],
    [Indicators.gust]: [0, 200],
    [Indicators.ugrd]: [-100, 100],
    [Indicators.vgrd]: [-100, 100],
};

export const from255 = (value: number, indicator: Indicators) =>
    (value * (dataExtremes[indicator][1] - dataExtremes[indicator][0])) / 255 + dataExtremes[indicator][0];

useDialog.js
import { calculateWind } from "~/core/wind/calculateWind";
import { from255 } from "./transform"

function pressureToAltitude(pressureHPa, seaLevelPressure = 1013.25) {
    const altitude = 44330 * (1 - Math.pow(pressureHPa / seaLevelPressure, 1 / 5.255));
    return Math.ceil(altitude);
}

export default function useDialog() {
    const gfsData = inject('gfsData')
    const chartData = shallowRef({})

    const indicators = computed(() => {
        return [
            {
                name: 'Температура'
            },
            {
                name: 'Давление'
            }
        ]
    })

    watch(gfsData, (data) => {
        const temp = { data: [], type: 'line', name: 'Температура', showSymbol: false }

        const uC = []
        const vC = []

        for (let i in data) {
            const value = data[i]
            const indicator = i.split("_")
            const h = pressureToAltitude(indicator[1].slice(0, -2))

            const transformvalue = from255(value, indicator[0])

            if (indicator[1].includes('mb'))
                switch (indicator[0]) {
                    case 'temp':
                        temp.data.push([transformvalue, h])
                        break;
                    case 'ugrd':
                        uC.push([transformvalue, h])
                        break;
                    case 'vgrd':
                        vC.push([transformvalue, h])
                        break;
                }
        }

        uC.map((u, i) => {
            const v = vC[i]

            const wind = calculateWind(u[0], v[0])

            return wind
        })


        chartData.value['temp'] = temp
    })

    return { chartData, indicators }
}

import { IControl, Map as MapLibreMap, MapMouseEvent } from "maplibre-gl";

interface RotateControlOptions {
    bearingSpeed?: number;
    pitchSpeed?: number;
}

export class RotateControl implements IControl {
    private _container: HTMLElement | null = null;
    private _map?: MapLibreMap;
    private _bearingSpeed: number;
    private _pitchSpeed: number;

    constructor(options?: RotateControlOptions) {
        this._bearingSpeed = options?.bearingSpeed || 0.5;
        this._pitchSpeed = options?.pitchSpeed || 0.3;
    }

    onAdd(map: MapLibreMap) {
        this._map = map;
        this._container = document.createElement("div");

        this._map.on("mousedown", this._onMouseDown);

        return this._container;
    }

    onRemove(map: MapLibreMap) {
        this._container?.parentNode?.removeChild(this._container);
        this._map = undefined;

        map.off("mousedown", this._onMouseDown);
    }

    private _onMouseDown = (evt: MapMouseEvent & Object) => {
        if (evt.originalEvent.button !== 2) return;

        evt.originalEvent.preventDefault();

        const onMouseMove = (evt: MouseEvent) => {
            if (!this._map) return;

            const newBearing =
                this._map.getBearing() + evt.movementX * this._bearingSpeed;
            const newPitch = this._map.getPitch() - evt.movementY * this._pitchSpeed;

            this._map.jumpTo({
                bearing: newBearing,
                pitch: Math.min(85, Math.max(0, newPitch)),
            });
        };

        const onMouseUp = () => {
            window.removeEventListener("mousemove", onMouseMove);
            window.removeEventListener("mouseup", onMouseUp);
        };

        window.addEventListener("mousemove", onMouseMove);
        window.addEventListener("mouseup", onMouseUp);
    };
}

useDeckGL.js
import { MapboxOverlay } from '@deck.gl/mapbox';
import { GeoJsonLayer, } from '@deck.gl/layers';
import colorer from '~/core/colorer';

const polygons = []

for (let i = 150; i <= 375; i += 25) {
    polygons.push(await import(`~/assets/${i}.json`))
}

const addZ = (coords, z) => {
    return typeof coords[0] === 'number'
        ? [coords[0], coords[1], z]
        : coords.map(c => addZ(c, z))
}

const hexToRgb = (hex) => {
    const string = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    const o = string ? {
        r: parseInt(string[1], 16),
        g: parseInt(string[2], 16),
        b: parseInt(string[3], 16)
    } : null;

    return [o.r, o.g, o.b, 100]
}

export default function useDeckGL({ map }) {

    const selectedHeight = inject('selectedHeight')
    const visibleDeckGL = inject('visibleDeckGL')

    map.on('load', () => {
        const processedData = computed(() => {
            const features = polygons
                .flatMap(c => c.features)
                .filter(f => f.properties.H === selectedHeight.value)

            return {
                type: 'FeatureCollection',
                features: features.map(f => ({
                    ...f,
                    geometry: {
                        ...f.geometry,
                        coordinates: addZ(f.geometry.coordinates, f.properties.H + 100)
                    }
                }))
            }
        })

        const layer = new GeoJsonLayer({
            id: `polygonDeckGL`,
            data: processedData.value,
            extruded: true,
            getFillColor: f => {
                return hexToRgb(colorer.paintSelf('m-t', f.properties.t))
            },
            getElevation: 10,
        })

        const overlay = new MapboxOverlay({
            interleaved: false,
            layers: [layer]
        })

        map.addControl(overlay);

        watch(visibleDeckGL, (visible) => {
            const layer = new GeoJsonLayer({
                id: `polygonDeckGL`,
                data: processedData.value,
                visible: visible,
                extruded: true,
                getFillColor: f => {
                    return hexToRgb(colorer.paintSelf('m-t', f.properties.t))
                },
                getElevation: 10,
            })

            overlay.setProps({ layers: [layer] })
        })

        watch(selectedHeight, (height) => {
            const layer = new GeoJsonLayer({
                id: `polygonDeckGL`,
                data: processedData.value,
                visible: visibleDeckGL.value,
                extruded: true,
                getFillColor: f => {
                    return hexToRgb(colorer.paintSelf('m-t', f.properties.t))
                },
                getElevation: 10,
            })

            overlay.setProps({ layers: [layer] })
        })
    })
}

useFill.js
import colorer from "~/core/colorer";

const polygons = []

for (let i = 150; i <= 375; i += 25) {
    polygons.push(await import(`~/assets/${i}.json`))
}

export default function usePolygons({ map }) {

    const visibleFill = inject('visibleFill')
    const selectedHeight = inject('selectedHeight')

    map.on('load', () => {

        polygons.forEach((item, i) => {
            map.addSource(`PolygonSource${item.name}`, {
                type: 'geojson',
                data: { ...item }
            });

            map.addLayer({
                id: `Polygon${item.name}`,
                type: 'fill',
                source: `PolygonSource${item.name}`,
                layout: { visibility: visibleFill.value ? 'visible' : 'none' },
                paint: {
                    'fill-opacity': 0.1,
                    "fill-color": ['interpolate',
                        ['linear'],
                        ['get', 't'],
                        ...colorer.getScheme('m-t')
                    ]
                }
            });
        })
    })

    watch(visibleFill, (value) => {
        if (selectedHeight.value == 125) {
            polygons.forEach((item) => {
                map.setLayoutProperty(
                    `Polygon${item.name}`,
                    'visibility',
                    value ? 'visible' : 'none'
                );

                map.setPaintProperty(
                    `Polygon${item.name}`,
                    'fill-opacity', 0.1)
            })
        }
        else {
            if (value) {
                polygons.forEach((item) => {
                    if (selectedHeight.value == item.name) {
                        map.setLayoutProperty(
                            `Polygon${item.name}`,
                            'visibility',
                            'visible'
                        );

                        map.setPaintProperty(
                            `Polygon${item.name}`,
                            'fill-opacity', 0.5)
                    }
                    else
                        map.setLayoutProperty(
                            `Polygon${item.name}`,
                            'visibility',
                            'none'
                        );
                })
            }
            else {
                polygons.forEach((item) => {
                    if (selectedHeight.value == item.name) {
                        map.setLayoutProperty(
                            `Polygon${item.name}`,
                            'visibility',
                            'none'
                        );
                    }

                })
            }
        }
    })

    watch(selectedHeight, (height) => {
        if (height == 125) {
            polygons.forEach((item) => {
                map.setLayoutProperty(
                    `Polygon${item.name}`,
                    'visibility',
                    visibleFill.value ? 'visible' : 'none'
                );

import maplibregl from "maplibre-gl";
import * as THREE from "three";

function calculateDistanceMercatorToMeters(from, to) {
    const mercatorPerMeter = from.meterInMercatorCoordinateUnits();
    const dEast = to.x - from.x;
    const dEastMeter = dEast / mercatorPerMeter;
    const dNorth = from.y - to.y;
    const dNorthMeter = dNorth / mercatorPerMeter;
    return { dEastMeter, dNorthMeter };
}


export default function useTube({ map }) {
    map.on('load', () => {
        const tubes = [
            { x: 93.029659, y: 56.024332, height: 300, },
            { x: 93.033493, y: 56.025403, height: 250 },

            { x: 92.898606, y: 55.970713, height: 250 },
            { x: 92.901071, y: 55.970737, height: 250 },

            { x: 93.094622, y: 56.109603, height: 250 },
            { x: 93.090299, y: 56.106921, height: 150 }
        ]

        tubes.forEach((tube, index) => {

            const sceneOrigin = new maplibregl.LngLat(tube.x, tube.y);

            const customLayer = {
                id: `3d-model ${index}`,
                type: "custom",
                renderingMode: "3d",
                onAdd: function (map, gl) {
                    try {
                        this.camera = new THREE.Camera();
                        this.scene = new THREE.Scene();
                        this.map = map;
                        this.renderer = new THREE.WebGLRenderer({
                            canvas: map.getCanvas(),
                            context: gl,
                            antialias: true,
                        });
                        this.renderer.autoClear = false;

                        const scene = new maplibregl.LngLat(tube.x, tube.y);
                        const modelLocation = new maplibregl.LngLat(tube.x, tube.y);

                        const sceneElevation = map.queryTerrainElevation(scene);
                        const model1Elevation = map.queryTerrainElevation(modelLocation);

                        const model1up = model1Elevation + (tube.height / 2) - sceneElevation;
                        const sceneOriginMercator = maplibregl.MercatorCoordinate.fromLngLat(scene);
                        const model1Mercator = maplibregl.MercatorCoordinate.fromLngLat(modelLocation);

                        const { dEastMeter: model1east, dNorthMeter: model1north } = calculateDistanceMercatorToMeters(sceneOriginMercator, model1Mercator);

                        const geometry = new THREE.CylinderGeometry(10, 20, tube.height, 16, 3);

                        const material = new THREE.MeshBasicMaterial({ color: 0xcccccc });

                        const tubeModel = new THREE.Mesh(geometry, material);

                        tubeModel.position.set(model1east, model1north, model1up);

                        tubeModel.rotation.x = Math.PI / 2

                        this.scene.add(tubeModel);

                    }

                    catch (e) {
                        console.log(e)
                    }
                },

                render: function (gl, args) {
                    const offsetFromCenterElevation = map.queryTerrainElevation(sceneOrigin) || 0;
                    const sceneOriginMercator = maplibregl.MercatorCoordinate.fromLngLat(sceneOrigin, offsetFromCenterElevation);

                    const sceneTransform = {
                        translateX: sceneOriginMercator.x,
                        translateY: sceneOriginMercator.y,
                        translateZ: sceneOriginMercator.z,
                        scale: sceneOriginMercator.meterInMercatorCoordinateUnits()
                    };

                    const m = new THREE.Matrix4().fromArray(args.defaultProjectionData.mainMatrix);
                    const l = new THREE.Matrix4()
                        .makeTranslation(sceneTransform.translateX, sceneTransform.translateY, sceneTransform.translateZ)
                        .scale(new THREE.Vector3(sceneTransform.scale, -sceneTransform.scale, sceneTransform.scale));

                    this.camera.projectionMatrix = m.multiply(l);
                    this.renderer.resetState();
                    this.renderer.render(this.scene, this.camera);
                    map.triggerRepaint();
                },
            };

            map.addLayer(customLayer)
        })
    })
}

useSiteDialog.ts
import type { CalendarDate } from "@internationalized/date";
import type { SeriesOption } from "echarts";
import type { MarkLineOption } from "echarts/types/src/component/marker/MarkLineModel.js";
import type { Feature } from "maplibre-gl";
import type { ShallowRef } from "vue";
import getDiff from "~/core/time/getDiff";
import type Indicator from "~/types/indicator";
import type SiteData from "~/types/siteData";
import SetDataService from "~/core/data/setDataService";

const setDialogDataService = new SetDataService()

export default function useSiteDialog() {
    const selectedFeature = inject<ShallowRef<Feature>>('selectedFeature')
    const dataGroupedBySite = inject<ComputedRef<any>>('dataGroupedBySite')
    const selectedIndicator = inject<ShallowRef<Indicator>>('selectedIndicator')
    const selectedDate = inject<ShallowRef<string>>('selectedDate')
    const selectedDates = inject<ShallowRef<{ start: CalendarDate, end: CalendarDate }>>('selectedDates')
    const indicators = inject<Indicator[]>('indicators')
    const loading = inject<ShallowRef<boolean>>('loading')!

    const timePeriod = shallowRef<{ start: CalendarDate, end: CalendarDate }>({ ...selectedDates!.value })

    const value = shallowRef<string[]>([selectedIndicator?.value.code || 'm-t'])

    const siteData = shallowRef<SiteData[]>([])

    const lastData = computed<Indicator[]>(() => {

        if (siteData.value.length == 0) {
            return []
        }

        const sums: any = {};
        const counts: any = {}

        indicators!.forEach(indicator => {
            sums[indicator.code] = 0;
            counts[indicator.code] = 0
        });

        siteData.value.forEach(obj => {
            indicators!.forEach(indicator => {
                if (obj[indicator.code]) {
                    sums[indicator.code] += obj[indicator.code];
                    counts[indicator.code]++
                }
            });
        });

        const averages: any = [];

        indicators!.forEach(indicator => {
            const avg = sums[indicator.code] / counts[indicator.code]
            const value = `${avg.toFixed(3)} ${indicator.units || ''}`
            if (avg)
                averages.push({ ...indicator, value })
        })

        return averages || []
    })

    const chartData = computed<SeriesOption[]>(() => {
        if (!selectedFeature?.value) {
            return []
        }

        const markLine: MarkLineOption = {
            silent: true,
            animation: false,
            symbol: 'none',
            label: {
                position: 'end',
                color: 'inherit'
            },
            data: [{ xAxis: selectedDate?.value, }]
        }

        const res: SeriesOption[] = value.value.map((code, index) => {
            const indicator = indicators?.find(i => i.code == code)
            const data = siteData.value.map((d: any) => {
                return [d.time, d[code || 0] || null]
            })
            return { data, type: 'line', name: indicator?.name, showSymbol: true, yAxisIndex: index, symbolSize: 2 }
        })

        res[0]!.markLine = markLine

        return res

    })

    watch(() => selectedFeature, (shallowFeature) => {
        const { properties } = shallowFeature!.value
        const id = properties.site
        const data = dataGroupedBySite?.value[id] || []

        siteData.value = data

    }, { immediate: true })

    watch(timePeriod, async (newTimePeriod) => {
        const begin = newTimePeriod.start.toString()
        const end = newTimePeriod.end.toString()
        const id = selectedFeature!.value.properties.site
        const codeDta = selectedFeature!.value.properties.codeDta
        const difference = getDiff(newTimePeriod.start, newTimePeriod.end)
        const timeInterval = difference < 5 ? 'hour' : 'day'

        loading.value = true

        const data = await setDialogDataService.getDataById({ begin, end, timeInterval, codeDta, id })
            .finally(() =>
                loading.value = false)

timeline.ts
import colorer from "~/core/colorer";
import getDiff from "~/core/time/getDiff"
import findDayChangeIndex from "~/scripts/searchDay"
import type Indicator from "~/types/indicator";
import type SelectedDates from "~/types/selectedDates";
import type { Point, Points, Timeline, TimelineMode, Transition } from "~/types/timeline";

export function createPoints(points: string[], backgrounds: any, indicator: Indicator): Points {
    return points.map((item) => {
        const background = backgrounds[item][indicator.code] || { color: "#fff", value: 0 }

        return {
            time: item,
            type: 'point',
            background: background.color,
            value: background.value
        }
    })
}

export function createBackgrounds(avg: any) {
    return Object.fromEntries(
        Object.entries(avg).map(([time, value]) => [time, colorer.paintObject(value)])
    )
}

export function setTransition(points: Points, transition: Transition[]) {
    let count = 0

    transition.forEach(item => {
        points.splice(item.newDayStartIndex + count, 0, { type: 'pointBorder', time: item.newDay || 'Не указано' })
        count++
    })
}

export function createTimeline({ borders, points, backgrounds, indicator }: { borders: SelectedDates, points: string[], backgrounds: any, indicator: Indicator }): Timeline {
    const transition: Transition[] = findDayChangeIndex(points)
    const timelineMode: TimelineMode = getDiff(borders.start, borders.end) < 5 ? 'hour' : 'day'
    const newPoints: Points = createPoints(points, backgrounds, indicator)

    if (timelineMode == 'hour')
        setTransition(newPoints, transition)

    return { mode: timelineMode, points: newPoints }
}

useTimeline.ts
import type { ShallowRef } from "vue";
import { createBackgrounds, createTimeline } from "./timeline";
import type Indicator from "~/types/indicator";
import type { CalendarDate } from "@internationalized/date";
import type { Point, Points, Timeline } from "~/types/timeline";


export default function useTimeline({ items, avg }: { items: Ref<string[]>, avg: any }) {
    const selectedDate = inject<ShallowRef<string | undefined>>('selectedDate')
    const selectedDates = inject<ShallowRef<{ start: CalendarDate, end: CalendarDate }>>('selectedDates')
    const selectedIndicator = inject<ShallowRef<Indicator>>('selectedIndicator')

    const itemsInPage: number = 15
    const selectedIndex = ref<number>(0)
    const page = ref<number>(0)

    const backgrounds = computed(() => {
        return createBackgrounds(avg.value)
    })

    const timeline = shallowRef<Timeline>({ mode: 'hour', points: [] })

    const pointsByPage = computed<Points>(() => {
        const from = itemsInPage * page.value;
        const to = from + itemsInPage;
        return timeline.value.points.slice(from, to);
    });

    const maxPage = computed<number>(() => {
        if (!items.value) return 0;
        return Math.ceil(timeline.value.points.length / itemsInPage);
    });

    const increasePage = () => {
        if (page.value + 1 < maxPage.value) {
            page.value += 1;
        }
    }

    const decreasePage = () => {
        if (page.value > 0) {
            page.value -= 1
        }
    }

    const setSelected = (item: Point, index: number) => {
        selectedIndex.value = index
    }

    const pressArrowLeft = () => {
        let index = selectedIndex.value - 1;
        let point = timeline.value.points[index];

        if (point && selectedDate?.value) {

            if (point.type == 'pointBorder') {
                index = selectedIndex.value - 2;
                point = timeline.value.points[index];
            }

            selectedIndex.value = index;
            page.value = Math.floor(index / itemsInPage);

            selectedDate.value = point?.time
        }
    };

    const pressArrowRight = () => {
        let index = selectedIndex.value + 1;
        let point = timeline.value.points[index];

        if (point && selectedDate?.value) {

            if (point.type == 'pointBorder') {
                index = selectedIndex.value + 2;
                point = timeline.value.points[index];
            }

            selectedIndex.value = index;
            page.value = Math.floor(index / itemsInPage);

            selectedDate.value = point?.time
        }
    }

    watch(items, (value) => {
        timeline.value = createTimeline({ points: value, borders: selectedDates!.value, backgrounds: backgrounds.value, indicator: selectedIndicator!.value })

        const finded = timeline.value.points.findIndex(item => item.time == selectedDate?.value)

        if (finded == -1 && timeline.value.points[0] && selectedDate?.value) {
            selectedDate.value = timeline.value.points[0].time
            page.value = 0;
        }

        if (finded > -1) {
            page.value = Math.floor(finded / itemsInPage);
            selectedIndex.value = finded + itemsInPage * page.value
        }

    }, { immediate: true })

    watch(() => selectedDates, () => {
        page.value = 0
    })

    return {
        page,
        maxPage,
        pointsByPage,
        selectedIndex,
        timeline,
        itemsInPage,
        selectedDates,
        pressArrowLeft,
        pressArrowRight,
        increasePage,
        decreasePage,
        setSelected
    }
}

avg.ts
import type Indicator from "~/types/indicator"
import type SiteData from "~/types/siteData"
import groupByHour from "../group/groupByHour"

export const avg = (setData: SiteData[], indicators: Indicator[]) => {
    const grouped = groupByHour<SiteData>(setData)

    const res: any = {}

    indicators.forEach(indicator => {
        const code = indicator.code

        let avg = 0

        for (let i in grouped) {
            if (!res[i])
                res[i] = {}

            const day = grouped[i]!.filter(g => g[code])

            const sum = day.reduce((acc, value) => {
                const v = value[code]!
                return acc + +v;
            }, 0);

            avg = sum / day.length || 0


            res[i][code] = avg
        }
    })

    return res
}

import colorizeByRange from "./colorize"
import colorize from "./colorizeSelf";

class Colorer {
    constructor({ schemes }) {
        this.schemes = schemes
    }

    register(key, schema) {
        this.schemes[key] = schema
    }

    paint(key, data) {
        const schema = this.schemes[key]
        const colored = colorizeByRange(data, schema);

        return colored
    }

    paintSelf(key, value) {
        const schema = this.schemes[key]
        return colorize(value, schema)
    }

    paintObject(data) {
        const res = {}

        for (let key in data) {
            const value = data[key]
            res[key] = { color: colorize(value, this.schemes[key]), value }
        }

        return res
    }

    getScheme(key) {
        return this.schemes[key] || [0, "#ffffff"]
    }
}

export default Colorer

colorize.js
export default function colorizeByRange(items, colorRanges, fieldValue = 'value') {
    return items.map(obj => {
        const value = obj[fieldValue];

        for (let i = 0; i < colorRanges.length; i += 2) {
            const rangeEnd = colorRanges[i];
            const color = colorRanges[i + 1];

            if (value <= rangeEnd || i === colorRanges.length - 2) {
                return {
                    ...obj,
                    color: color
                };
            }
        }

        return obj;
    });
}

colorizeSelf.js
export default function colorize(value, colorScheme) {
    if (value <= colorScheme[0]) {
        return colorScheme[1];
    }

    for (let i = 0; i < colorScheme.length; i += 2) {
        const threshold = colorScheme[i];
        if (value < threshold) {
            return colorScheme[i - 1];
        }
    }

    return colorScheme[colorScheme.length - 1];
}

export default class DataService {

    setService: SetService

        this.setDataService = setDataService
        this.setService = setService
    }

    async getData(options: any) {
        const { begin, end, dta } = options

        const beginDate = begin.toString()
        const endDate = end.toString()

        const difference = getDiff(begin, end)
        const timeInterval = difference < 5 ? 'hour' : 'day'

        const data = await this.setDataService.getData({ begin: beginDate, end: endDate, timeInterval, dta })
        return data
    }

    async getDta() {
        const data = await this.setService.getSets()
        return data
    }
}

                url.searchParams.set(option, options[option])
        }
    }

    async getDataById({ begin, end, timeInterval, codeDta, id }: { begin: string, end: string, timeInterval: string, codeDta: string, id: string }) {

        const url = new URL(`${codeDta}/data/archive`, this.URL)

        const options = {
            'time_begin': `${formateDate(begin)} 00:00:00`,
            'time_end': `${formateDate(end)} 23:59:00`,
            'uid': 'khlnn3o37xah7pj2',
            'time_interval': timeInterval,
            'sites': id
        }

        this.setParams(options, url)

        const promise = await fetch(url.toString()).then((res) => res.json())

        return promise.data
    }

    async getData({ dta, begin, end, timeInterval }: { dta: Dta[], begin: string, end: string, timeInterval: string }) {

        const promises = dta.map(set => {

            const url = new URL(`${set.code}/data/archive`, this.URL)

            const options = {
                'time_begin': `${formateDate(begin)} 00:00:00`,
                'time_end': `${formateDate(end)} 23:59:00`,
                'uid': 'khlnn3o37xah7pj2',
                'time_interval': timeInterval
            }

            this.setParams(options, url)

            const promise = fetch(url.toString()).then((res) => res.json())

            return promise
        })

        const res = await Promise.all(promises).then((res) => {
            const arrays: any = []

            res.forEach(r => arrays.push(...r.data))

        })

const createSiteFeature = (site: Site, data: SiteData): Feature => {
    return {
        type: "Feature",
        properties: {
            ...site,
            ...data
        },
        geometry: {
            type: "Point",
            coordinates: [site.geom_x, site.geom_y],
        },
    };
};

const createSiteFeatureCollection = (dta: Dta[], data: SiteData[]): FeatureCollectionsMap => {
    const sites: Site[] = dta.flatMap(({ sites }) => sites);

    const groupedByHour = groupByHour<SiteData>(data)

    const res: FeatureCollectionsMap = {}

    for (const time in groupedByHour) {

        const featuresCollection: FeatureCollection = {
            type: "FeatureCollection",
            features: []
        }

        const features: Feature[] = []

        sites.forEach(site => {
            const filtered = groupedByHour[time]?.find(i => i.site == site.id)
            if (filtered)
                features.push(createSiteFeature(site, filtered))

        })

        featuresCollection.features = features

        res[time] = featuresCollection
    }

    return res
};

export { createSiteFeatureCollection };

export default function groupByHour(data, field) {
    const grouped = {};

    data.forEach(item => {
        const dateTime = item[field];

        const hourKey = dateTime

        if (!grouped[hourKey]) {
            grouped[hourKey] = [];
        }

        grouped[hourKey].push(item);
    });

    return grouped;
}

export default function groupByHour<T>(data: any[]) {
    const grouped: Record<string, T[]> = {};

    data.forEach(item => {
        const dateTime = item.time;

        const hourKey = dateTime.substring(0, 19);

        if (!grouped[hourKey]) {
            grouped[hourKey] = [];
        }

        grouped[hourKey].push(item);
    });

    return grouped;
}

import { CalendarDate, } from '@internationalized/date'
export default function getDiff(b: CalendarDate, e: CalendarDate): number {
    const begin = b.toString()
    const end = e.toString()

    const diff = new Date(end).getTime() - new Date(begin).getTime();
    return Math.round(diff / (1000 * 3600 * 24));
}

export const calculateWind = (u: number, v: number) => {
    u = from255(u, Indicators.ugrd);
    v = from255(v, Indicators.vgrd);

        direction: Math.atan(v / u),
    };

formate.js
export default function formateDate(d) {
    const date = new Date(d);
    let dd = date.getDate();
    let mm = date.getMonth() + 1;
    let yy = date.getFullYear();

    if (dd < 10) dd = "0" + dd;
    if (mm < 10) mm = "0" + mm;
    if (yy < 10) yy = "0" + yy;

    return yy + "-" + mm + "-" + dd;
}

function displayMultiPolygon(multiPolygonGeometry, map) {
    const geometry = new THREE.BufferGeometry();
    const vertices = [];
    const indices = [];
    let vertexCount = 0;

    // Обрабатываем все полигоны
    multiPolygonGeometry.features[0].geometry.coordinates.forEach(polygonGroup => {
        polygonGroup.forEach(polygon => {
            // Добавляем вершины для каждого полигона
            polygon.forEach(coord => {

                const point = map.project([coord[0], coord[1]]);

                point.x = point.x * 10
                point.y = point.y * 10

                vertices.push(
                    point.x,
                    point.y,
                    0
                );

                console.log(vertices)
            });

            for (let i = 1; i < polygon.length - 1; i++) {
                indices.push(
                    vertexCount,
                    vertexCount + i,
                    vertexCount + i + 1
                );
            }

            vertexCount += polygon.length;
        });
    });

    geometry.setIndex(indices);
    geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));

    const material = new THREE.MeshBasicMaterial({
        color: 0x00ff00,
        side: THREE.DoubleSide,
        transparent: true,
        opacity: 0.7
    });

    const mesh = new THREE.Mesh(geometry, material)

    return mesh
}

export default displayMultiPolygon

searchDay.ts
import type { Transition } from "~/types/timeline";

function findDayChangeIndex(datesArray: string[]): Transition[] {
    if (datesArray.length <= 1) return [];

    const transitions: Transition[] = [];

    const datesOnly = datesArray.map(dateStr => dateStr.split(' ')[0]);

    for (let i = 1; i < datesOnly.length; i++) {
        if (datesOnly[i] !== datesOnly[i - 1]) {
            transitions.push({
                oldDayEndIndex: i - 1,
                newDayStartIndex: i,
                oldDay: datesOnly[i - 1] || null,
                newDay: datesOnly[i] || null,
                transitionPoint: i
            });
        }
    }

    return transitions;
}

export default findDayChangeIndex
